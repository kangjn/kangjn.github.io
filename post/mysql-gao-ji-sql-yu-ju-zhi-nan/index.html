MySQL 高级 SQL 语句指南

MySQL 高级 SQL 语句指南

  1. 窗口函数 (Window Functions)
    -- 排名：RANK 有并列跳号，DENSE_RANK 不跳号，ROW_NUMBER 不并列
    SELECT
    name, department, salary,
    RANK() OVER (PARTITION BY department ORDER BY salary DESC) AS rnk,
    DENSE_RANK() OVER (PARTITION BY department ORDER BY salary DESC) AS dense_rnk,
    ROW_NUMBER() OVER (PARTITION BY department ORDER BY salary DESC) AS row_num
    FROM employees;

-- 累计求和
SELECT
order_date, amount,
SUM(amount) OVER (ORDER BY order_date ROWS UNBOUNDED PRECEDING) AS running_total
FROM orders;

-- 移动平均（前后各2行）
SELECT
order_date, amount,
AVG(amount) OVER (ORDER BY order_date ROWS BETWEEN 2 PRECEDING AND 2 FOLLOWING) AS moving_avg
FROM orders;

-- LAG / LEAD：取前一行/后一行的值
SELECT
month, revenue,
LAG(revenue, 1) OVER (ORDER BY month) AS prev_month,
revenue - LAG(revenue, 1) OVER (ORDER BY month) AS growth
FROM monthly_sales;

-- NTILE：将数据等分为 N 组
SELECT name, salary,
NTILE(4) OVER (ORDER BY salary DESC) AS quartile
FROM employees;

  1. CTE (公用表表达式)
    -- 基础 CTE
    WITH dept_stats AS (
    SELECT department_id, AVG(salary) AS avg_salary
    FROM employees
    GROUP BY department_id
    )
    SELECT e.name, e.salary, d.avg_salary
    FROM employees e
    JOIN dept_stats d ON e.department_id = d.department_id
    WHERE e.salary > d.avg_salary;

-- 递归 CTE：组织架构层级查询
WITH RECURSIVE org_tree AS (
-- 锚点：顶层（无上级）
SELECT id, name, manager_id, 1 AS level, CAST(name AS CHAR(500)) AS path
FROM employees WHERE manager_id IS NULL
UNION ALL
-- 递归
SELECT e.id, e.name, e.manager_id, t.level + 1,
CONCAT(t.path, ' → ', e.name)
FROM employees e
JOIN org_tree t ON e.manager_id = t.id
)
SELECT * FROM org_tree ORDER BY path;

  1. 高级 JOIN 技巧
    -- 自连接：找同部门薪资更高的员工
    SELECT a.name, a.salary, b.name AS higher_earner
    FROM employees a
    JOIN employees b ON a.department_id = b.department_id AND b.salary > a.salary;

-- 交叉连接生成日历（补零）
WITH RECURSIVE calendar AS (
SELECT '2026-01-01' AS dt
UNION ALL
SELECT DATE_ADD(dt, INTERVAL 1 DAY) FROM calendar WHERE dt < '2026-12-31'
)
SELECT c.dt, COALESCE(SUM(o.amount), 0) AS daily_sales
FROM calendar c
LEFT JOIN orders o ON DATE(o.order_date) = c.dt
GROUP BY c.dt;

-- LATERAL JOIN（MySQL 8.0.14+）：每组取 Top N
SELECT d.name, top3.*
FROM departments d
CROSS JOIN LATERAL (
SELECT e.name AS emp_name, e.salary
FROM employees e
WHERE e.department_id = d.id
ORDER BY e.salary DESC
LIMIT 3
) top3;

  1. 条件表达式与聚合

-- 行转列（PIVOT）
SELECT
student_name,
MAX(CASE WHEN subject = '数学' THEN score END) AS math,
MAX(CASE WHEN subject = '英语' THEN score END) AS english,
MAX(CASE WHEN subject = '物理' THEN score END) AS physics
FROM scores
GROUP BY student_name;

-- 列转行（UNPIVOT）
SELECT id, 'Q1' AS quarter, q1_sales AS sales FROM sales_data
UNION ALL
SELECT id, 'Q2', q2_sales FROM sales_data
UNION ALL
SELECT id, 'Q3', q3_sales FROM sales_data;

-- GROUPING SETS 模拟（MySQL 不原生支持）
SELECT department, job_title, SUM(salary)
FROM employees
GROUP BY department, job_title
UNION ALL
SELECT department, NULL, SUM(salary)
FROM employees
GROUP BY department
UNION ALL
SELECT NULL, NULL, SUM(salary)
FROM employees;

  1. 存储过程与游标
    DELIMITER //
    CREATE PROCEDURE generate_report(IN dept_id INT, OUT total DECIMAL(12,2))
    BEGIN
    DECLARE done INT DEFAULT FALSE;
    DECLARE emp_salary DECIMAL(10,2);
    DECLARE cur CURSOR FOR
    SELECT salary FROM employees WHERE department_id = dept_id;
    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;

    SET total = 0;
    OPEN cur;
    read_loop: LOOP
    FETCH cur INTO emp_salary;
    IF done THEN LEAVE read_loop; END IF;
    SET total = total + emp_salary;
    END LOOP;
    CLOSE cur;
    END //
    DELIMITER ;

-- 调用
CALL generate_report(10, @total);
SELECT @total;

  1. 高级子查询
    -- EXISTS vs IN（大数据集 EXISTS 通常更快）
    SELECT * FROM customers c
    WHERE EXISTS (
    SELECT 1 FROM orders o
    WHERE o.customer_id = c.id AND o.amount > 1000
    );

-- 相关子查询：每个部门薪资最高的人
SELECT * FROM employees e
WHERE salary = (
SELECT MAX(salary) FROM employees
WHERE department_id = e.department_id
);

-- ALL / ANY
SELECT name, salary FROM employees
WHERE salary > ALL (
SELECT salary FROM employees WHERE department_id = 5
);

  1. 性能优化写法

-- 覆盖索引查询：只查索引列，避免回表
SELECT department_id, COUNT(*) FROM employees
WHERE department_id BETWEEN 1 AND 10
GROUP BY department_id;

-- 避免 SELECT *，只取需要的列
-- 避免在 WHERE 中对列做函数操作（会导致索引失效）
-- ✗ WHERE YEAR(order_date) = 2026
-- ✓ WHERE order_date >= '2026-01-01' AND order_date < '2027-01-01'

-- 分页优化（深分页）
-- ✗ SELECT * FROM orders LIMIT 1000000, 20;
-- ✓ 使用游标分页
SELECT * FROM orders
WHERE id > 1000000 -- 上一页最后一条的 ID
ORDER BY id
LIMIT 20;

-- USING 代替 ON（同名字段时更简洁）
SELECT * FROM orders JOIN customers USING (customer_id);

    -- UPSERT（存在则更新，不存在则插入）
    INSERT INTO product_stats (product_id, view_count)
    VALUES (101, 1)
    ON DUPLICATE KEY UPDATE view_count = view_count + 1;

-- 生成序列号
SELECT
@rownum := @rownum + 1 AS seq,
e.*
FROM employees e, (SELECT @rownum := 0) r
ORDER BY e.hire_date;

-- GROUP_CONCAT：将多行合并为一行
SELECT department_id,
GROUP_CONCAT(name ORDER BY salary DESC SEPARATOR ', ') AS members
FROM employees
GROUP BY department_id;

-- 窗口函数实现分组内百分比
SELECT
name, department, salary,
ROUND(salary / SUM(salary) OVER (PARTITION BY department) * 100, 2) AS pct
FROM employees;

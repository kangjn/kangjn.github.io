  1. EXPLAIN 执行计划解读
    EXPLAIN SELECT * FROM orders WHERE customer_id = 100;
    EXPLAIN ANALYZE SELECT ...; -- MySQL 8.0.18+，含实际执行时间
    EXPLAIN FORMAT=JSON SELECT ...; -- 更详细的信息
    EXPLAIN FORMAT=TREE SELECT ...; -- 树形执行计划

各字段含义
+----+-------------+-------+------+---------------+------+---------+------+------+-------+
| id | select_type | table | type | possible_keys | key | key_len | ref | rows | Extra |
+----+-------------+-------+------+---------------+------+---------+------+------+-------+

type（访问类型）— 从好到差：
system > const > eq_ref > ref > range > index > ALL

type 含义 示例
const 主键/唯一索引等值查询，最多一行 WHERE id = 1
eq_ref JOIN 时驱动表每一行都通过主键/唯一索引匹配 JOIN ... ON a.id = b.a_id
ref 非唯一索引等值查询 WHERE name = 'Tom'
range 索引范围扫描 WHERE id BETWEEN 1 AND 100
index 全索引扫描（只遍历索引树） SELECT indexed_col FROM t
ALL 全表扫描 🔴 SELECT * FROM t WHERE no_index_col = 1

Extra 常见值：

Extra 含义 好坏
Using index 覆盖索引，无需回表 ✅
Using where Server 层过滤 ⚠️
Using temporary 使用临时表 🔴
Using filesort 额外排序 🔴
Using index condition 索引下推 (ICP) ✅
Using join buffer JOIN 缓冲 ⚠️

  1. 索引优化策略

2.1 最左前缀原则

-- 联合索引 idx_a_b_c (a, b, c)

-- ✅ 命中索引
WHERE a = 1
WHERE a = 1 AND b = 2
WHERE a = 1 AND b = 2 AND c = 3
WHERE a = 1 AND b > 2 AND c = 3 -- a 等值 + b 范围（c 断了）

-- ❌ 无法命中
WHERE b = 2 -- 跳过了 a
WHERE c = 3 -- 跳过了 a, b
WHERE b = 2 AND c = 3 -- 跳过了 a

2.2 索引失效场景

-- ❌ 对索引列做函数/运算
WHERE YEAR(create_time) = 2026 -- 索引失效
WHERE create_time + 1 > '2026-01-01' -- 索引失效
-- ✅ 改写
WHERE create_time >= '2026-01-01' AND create_time < '2027-01-01'

-- ❌ 隐式类型转换
WHERE phone = 13800138000 -- phone 是 VARCHAR，传了数字 → 全表扫描
-- ✅
WHERE phone = '13800138000'

-- ❌ LIKE 左模糊
WHERE name LIKE '%Tom' -- 索引失效
-- ✅ 右模糊可以
WHERE name LIKE 'Tom%'

-- ❌ OR 连接非索引列
WHERE indexed_col = 1 OR no_index_col = 2 -- 可能全表扫描
-- ✅ 改用 UNION
SELECT * FROM t WHERE indexed_col = 1
UNION
SELECT * FROM t WHERE no_index_col = 2

-- ❌ NOT / != / <> / IS NOT NULL（部分场景失效）
-- ❌ 使用 SELECT *（无法使用覆盖索引）

2.3 索引设计原则

-- 1. 区分度高的列放前面
-- user_id（高区分度）放联合索引前面
ALTER TABLE orders ADD INDEX idx_user_status (user_id, status);

-- 2. 覆盖索引：把查询需要的列都放进索引
-- 查询只用 product_id 和 price
ALTER TABLE products ADD INDEX idx_cover (category_id, price, product_id);
SELECT product_id, price FROM products WHERE category_id = 10; -- Using index ✅

-- 3. 前缀索引：长字符串只取前 N 个字符
ALTER TABLE users ADD INDEX idx_email (email(10));
-- 验证选择性：
SELECT
COUNT(DISTINCT LEFT(email, 10)) / COUNT() AS sel_10,
COUNT(DISTINCT LEFT(email, 15)) / COUNT() AS sel_15,
COUNT(DISTINCT email) / COUNT(*) AS sel_full
FROM users;

-- 4. 索引合并（Index Merge）：MySQL 可能合并多个单列索引
-- 但不如直接建联合索引高效

  1. 慢查询分析

3.1 开启慢查询日志

-- 查看当前配置
SHOW VARIABLES LIKE 'slow_query%';
SHOW VARIABLES LIKE 'long_query_time';

-- 开启
SET GLOBAL slow_query_log = ON;
SET GLOBAL long_query_time = 1; -- 超过 1 秒记录
SET GLOBAL log_queries_not_using_indexes = ON; -- 记录未使用索引的查询

-- 查看慢查询条数
SHOW GLOBAL STATUS LIKE 'Slow_queries';

3.2 使用 mysqldumpslow 分析

按查询时间排序，取前 10 条

mysqldumpslow -s t -t 10 /var/log/mysql/slow.log

按出现次数排序

mysqldumpslow -s c -t 10 /var/log/mysql/slow.log

参数：-s 排序方式 (t=时间, c=次数, l=锁时间, r=行数)

3.3 SHOW PROFILE 分析各阶段耗时

SET profiling = 1;

SELECT * FROM orders WHERE customer_id = 100 ORDER BY create_time DESC LIMIT 20;

SHOW PROFILES; -- 查看所有查询 ID
SHOW PROFILE FOR QUERY 1; -- 查看详细阶段
SHOW PROFILE CPU, BLOCK IO FOR QUERY 1; -- 含 CPU 和 IO

  1. 查询重写技巧

4.1 子查询 → JOIN

-- ❌ 相关子查询（逐行执行，慢）
SELECT * FROM orders o
WHERE o.amount > (
SELECT AVG(amount) FROM orders WHERE customer_id = o.customer_id
);

-- ✅ 先算好平均值再 JOIN
SELECT o.* FROM orders o
JOIN (
SELECT customer_id, AVG(amount) AS avg_amt
FROM orders GROUP BY customer_id
) t ON o.customer_id = t.customer_id AND o.amount > t.avg_amt;

4.2 OR → UNION

-- ❌ OR 可能导致全表扫描
SELECT * FROM products WHERE category_id = 5 OR brand_id = 10;

-- ✅ 分开查再合并（各自走索引）
SELECT * FROM products WHERE category_id = 5
UNION ALL
SELECT * FROM products WHERE brand_id = 10 AND category_id != 5;

4.3 分页优化

-- ❌ 深分页（扫描 100 万行丢弃，只取 20 行）
SELECT * FROM orders ORDER BY id LIMIT 1000000, 20;

-- ✅ 延迟关联（先在索引上定位 ID，再回表）
SELECT o.* FROM orders o
JOIN (SELECT id FROM orders ORDER BY id LIMIT 1000000, 20) t
ON o.id = t.id;

-- ✅ 游标分页（最优，需要上一页最后一条的 ID）
SELECT * FROM orders WHERE id > 1000000 ORDER BY id LIMIT 20;

-- ✅ 业务层限制：禁止跳到第 N 页，只允许 "加载更多"

4.4 COUNT 优化

-- COUNT() vs COUNT(1) vs COUNT(col)
-- COUNT() = COUNT(1)：统计所有行，包括 NULL
-- COUNT(col)：统计 col 不为 NULL 的行

-- ✅ 如果不需要精确值，用近似值
SHOW TABLE STATUS LIKE 'orders'; -- Rows 字段是估算值

-- ✅ 用汇总表缓存计数
CREATE TABLE table_counts (
table_name VARCHAR(64) PRIMARY KEY,
row_count BIGINT
);
-- 在业务层维护增减

4.5 UPDATE/DELETE 优化

-- ❌ 大事务：一次性更新 100 万行
UPDATE orders SET status = 'archived' WHERE create_time < '2024-01-01';

-- ✅ 分批执行
REPEAT
UPDATE orders SET status = 'archived'
WHERE create_time < '2024-01-01' AND status != 'archived'
LIMIT 5000;
-- 检查 ROW_COUNT()
UNTIL ROW_COUNT() = 0 END REPEAT;

-- 用存储过程封装
DELIMITER //
CREATE PROCEDURE batch_update()
BEGIN
DECLARE affected INT DEFAULT 1;
WHILE affected > 0 DO
UPDATE orders SET status = 'archived'
WHERE create_time < '2024-01-01' AND status != 'archived'
LIMIT 5000;
SET affected = ROW_COUNT();
DO SLEEP(0.1); -- 降低锁压力
END WHILE;
END //
DELIMITER ;

  1. 索引监控与诊断

-- 查看索引使用情况（MySQL 8.0）
SELECT * FROM sys.schema_unused_indexes; -- 未使用的索引
SELECT * FROM sys.schema_redundant_indexes; -- 冗余索引
SELECT * FROM sys.schema_index_statistics ORDER BY rows_selected DESC;

-- 查看当前正在执行的查询
SHOW PROCESSLIST;
-- 或
SELECT * FROM information_schema.PROCESSLIST WHERE command != 'Sleep' ORDER BY time DESC;

-- 锁等待分析（MySQL 8.0）
SELECT * FROM performance_schema.data_lock_waits;
SELECT * FROM sys.innodb_lock_waits;

-- 表的索引信息
SHOW INDEX FROM orders;
SELECT * FROM information_schema.STATISTICS
WHERE TABLE_NAME = 'orders' ORDER BY SEQ_IN_INDEX;

个税计算公式

2026-06-04 4 min read

1、查表法
SELECT ISNULL(T.CurrentMonthTax, 0) AS CurrentMonthTax
FROM (VALUES(0)) AS Dummy(DefaultVal)
LEFT JOIN (
SELECT
CASE
WHEN CumulativeTax - PreviousCumulativeTax < 0 THEN 0
ELSE CumulativeTax - PreviousCumulativeTax
END AS CurrentMonthTax
FROM (
SELECT
c.EmpID,
c.YYMM,
c.CumTaxableIncome,
CASE
WHEN c.CumTaxableIncome <= 0 THEN 0
ELSE (c.CumTaxableIncome * (t.TaxRate / 100.0) - t.TakeOut)
END AS CumulativeTax,
LAG(
CASE
WHEN c.CumTaxableIncome <= 0 THEN 0
ELSE (c.CumTaxableIncome * (t.TaxRate / 100.0) - t.TakeOut)
END, 1, 0
) OVER (PARTITION BY c.EmpID ORDER BY c.YYMM) AS PreviousCumulativeTax
FROM (
SELECT
s.EmpID,
s.EmpNo,
s.YYMM,
s.Salary,
s.MonthNum,
s.CumSalary,
s.BasicDeduction,
d.CumSpecialDeduction,
(s.CumSalary - s.BasicDeduction - ISNULL(d.CumSpecialDeduction, 0) * s.MonthNum) AS CumTaxableIncome
FROM (
SELECT
EmpID,
EmpNo,
YYMM,
Salary,
MonthNum,
SUM(Salary) OVER (ORDER BY YYMM ROWS UNBOUNDED PRECEDING) AS CumSalary,
5000 * MonthNum AS BasicDeduction
FROM (
SELECT
EmpID,
EmpNo,
YYMM,
PayNo72 AS Salary,
ROW_NUMBER() OVER (ORDER BY YYMM) AS MonthNum
FROM PayMonthData
WHERE EmpID = @EmpID
AND YYMM <= @YYMM AND YYMM >= CAST(CAST(LEFT(@YYMM , 4) AS INT) - 1 AS CHAR(4)) + '12'
) AS SalaryHistory
) AS s
LEFT JOIN (
SELECT TOP 1
Field001,
Field003 AS CumSpecialDeduction
FROM CU003101Data
WHERE Field001 = (SELECT empno FROM PerEmployee WHERE EmpID =@EmpID)
AND Field004 <DATEADD(DAY, 1, EOMONTH(CONVERT(DATE,'202604' + '01')))
ORDER BY Field004 DESC
) AS d ON s.EmpNo = d.Field001
) AS c
OUTER APPLY (
-- 匹配税率表
SELECT TaxRate, TakeOut
FROM PayInComeTax
WHERE c.CumTaxableIncome >= IncludeBegin AND c.CumTaxableIncome <= IncludeEnd
) t
) AS TaxCalculation
WHERE YYMM = @YYMM
) AS T ON 1 = 1

2、直接算
DECLARE @YYMM VARCHAR(6) = '202302';
DECLARE @EmpID VARCHAR(20) = '张三';

WITH MonthlyData AS (
SELECT
EmpID,
YYMM,
payno69 - ISNULL(payno79, 0) AS NetSalary, -- 扣除社保公积金后金额
ISNULL(payno_special, 0) AS SpecialDeduction, -- 专项附加扣除（替换实际字段）
ROW_NUMBER() OVER (PARTITION BY EmpID ORDER BY YYMM) AS MonthNo
FROM PayMonthData
WHERE EmpID = @EmpID
AND YYMM >= LEFT(@YYMM, 4) + '01' -- 同年1月起累计
AND YYMM <= @YYMM
),
Cumulative AS (
SELECT
m.EmpID,
m.YYMM,
m.NetSalary,
m.MonthNo,
SUM(c.NetSalary) AS CumSalary,
SUM(c.SpecialDeduction) AS CumSpecial,
SUM(c.NetSalary) - SUM(c.SpecialDeduction) - 5000 * m.MonthNo AS TaxableIncome
FROM MonthlyData m
JOIN MonthlyData c ON m.EmpID = c.EmpID AND c.YYMM >= c.YYMM
GROUP BY m.EmpID, m.YYMM, m.NetSalary, m.MonthNo
),
TaxCalc AS (
SELECT
*,
CASE
WHEN TaxableIncome <= 0 THEN 0
WHEN TaxableIncome <= 36000 THEN TaxableIncome * 0.03
WHEN TaxableIncome <= 144000 THEN TaxableIncome * 0.10 - 2520
WHEN TaxableIncome <= 300000 THEN TaxableIncome * 0.20 - 16920
WHEN TaxableIncome <= 420000 THEN TaxableIncome * 0.25 - 31920
WHEN TaxableIncome <= 660000 THEN TaxableIncome * 0.30 - 52920
WHEN TaxableIncome <= 960000 THEN TaxableIncome * 0.35 - 85920
ELSE TaxableIncome * 0.45 - 181920
END AS CumTax
FROM Cumulative
)
SELECT
EmpID,
YYMM,
NetSalary AS 本月工资,
CumSalary AS 累计工资,
TaxableIncome AS 累计应纳税所得额,
CumTax AS 累计应扣税,
CumTax - ISNULL(LAG(CumTax) OVER (PARTITION BY EmpID ORDER BY YYMM), 0) AS 本月应扣个税
FROM TaxCalc
ORDER BY YYMM;
